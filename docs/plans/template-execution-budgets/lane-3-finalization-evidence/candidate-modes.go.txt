func finalRestrictedSetup(f executionWorkload,html bool)(*jet.Template,error){
 sources:=map[string]string{"workload":f.Source};for name,source:=range f.Templates{sources[name]=source}
 s,err:=jet.NewRestrictedSet(context.Background(),sources,jet.RestrictedOptions{Text:!html,Execution:finalExecutionOptions(),Input:jet.InputOptions{MaxBytes:1<<20,MaxItems:20000,MaxSteps:1000000,MaxDepth:64}})
 if err!=nil{return nil,err};return s.GetTemplate("workload")
}
